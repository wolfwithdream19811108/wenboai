function makeDateUrl($url,$startTime,$endTime){ //今日 $todayTime = time(); $today = date('Y-m-d',$todayTime); $todayString = "今日"; $todayClass = ''; if($startTime==$today&&$endTime==$today){ $todayClass = 'disabled'; } $todayString = "今日"; //昨日 $yesterday = date('Y-m-d',$todayTime-86400); $yesterdayString = "昨日"; $yesterdayClass = ''; if($startTime==$yesterday&&$endTime==$yesterday){ $yesterdayClass = 'disabled'; } $yesterdayString = "昨日"; //上一日 $prevStartDay = date('Y-m-d',strtotime($startTime)-86400); $prevEndDay = date('Y-m-d',strtotime($endTime)-86400); $prevDayString = "上一日"; //下一日 $nextStartDay = date('Y-m-d',strtotime($startTime)+86400); $nextEndDay = date('Y-m-d',strtotime($endTime)+86400); $nextClass = ''; if(strtotime($nextStartDay)>strtotime($today)||strtotime($nextEndDay)>strtotime($today)){ $nextClass = 'disabled'; } $nextDayString = "下一日"; //当月 $monthStartDay = date("Y-m-01",$todayTime); $monthEndDay = $today; //$monthDayString = "当月"; $monthClass = ''; if($monthStartDay==$startTime&&$monthEndDay==$endTime){ $monthClass = 'disabled'; } $monthDayString = "当月"; //上一月 $prevMonthStart = date('Y-m-01',strtotime("-1 month",strtotime($startTime))); $prevMonthEnd = date('Y-m-t',strtotime("-1 month",strtotime($startTime))); $prevMonthString = "上一月"; //下一月 $nextMonthStart = date('Y-m-01',strtotime("+1 month",strtotime($startTime))); $nextMonthEnd = date('Y-m-t',strtotime("+1 month",strtotime($startTime))); if(date('m',strtotime($nextMonthEnd))==date('m')){ $nextMonthEnd = $today; } $nextMonthClass = ''; if(strtotime($nextMonthStart)>strtotime($monthStartDay)){ $nextMonthClass = 'disabled'; } $nextMonthString = "下一月"; return sprintf('%s%s%s%s%s%s%s',$prevMonthString,$prevDayString,$monthDayString,$yesterdayString,$todayString,$nextDayString,$nextMonthString); }
